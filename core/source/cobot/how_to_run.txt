# Run your environment and move to the "rsl_rl" directory.
source isaac_sim51_lab/bin/activate
cd IsaacLab/scripts/reinforcement_learning/rsl_rl/

## To train
# With palm contact
python train.py --task Isaac-Lift-Cylinder-Cobot-v0 --num_envs 4096 --max_iterations 7000 --headless \
    --video --video_length 3000 --video_interval 90000 --enable_cameras
# without palm contact
python train.py --task Isaac-Lift-Cylinder2-Cobot-v0 --num_envs 4096 --max_iterations 7000 --headless \
    --video --video_length 3000 --video_interval 90000 --enable_cameras

## To play trained model

# move contents of the "trained_models" folder into "~/IsaacLab/scripts/reinforcement_learning/rsl_rl/logs/rsl_rl/" directory.

# With palm contact
python play.py --task Isaac-Lift-Cylinder-Cobot-v0 --num_envs 8 --checkpoint ./logs/rsl_rl/cobot_cylinder_ur5e/2026-07-12_08-20-44/model_5350.pt \
    --video --video_length 3000 --video_interval 90000 --enable_cameras
# without palm contact
python play.py --task Isaac-Lift-Cylinder2-Cobot-v0 --num_envs 8 --checkpoint ./logs/rsl_rl/cobot_cylinder2_ur5e/2026-07-13_22-29-47/model_6800.pt \
    --video --video_length 3000 --video_interval 90000 --enable_cameras

## Notes on video recording
# --video_length is the clip length in SIMULATION STEPS (cobot step_dt = 0.02 s, so 3000 steps = 60 s).
# --video_interval is the steps between clip START times (90000 steps = 30 min of sim time).
# A 1-minute clip is recorded immediately at simulation start, then every interval.
# Videos are written to <cwd>/logs/rsl_rl/<experiment_name>/<run>/videos/{train,play}/.
# Both values are input variables: adjust per task and env timestep (steps = seconds / step_dt).
